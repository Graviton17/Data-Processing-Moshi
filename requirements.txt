# or "torchvision has no attribute 'extension' / circular import").

torchaudio==2.6       # resampling; version must equal torch's

# pyannote >= 4.0 is required: 3.x calls hf_hub_download(use_auth_token=...),
# a kwarg removed in huggingface_hub 1.0. 4.x uses the modern token= API and
# reads HF_TOKEN from the environment. (No hub cap needed; 4.x supports hub 1.x.)
pyannote.audio>=4.0   # diarization + channel-purity re-diarization
